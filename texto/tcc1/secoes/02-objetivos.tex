\section{Objetivos}

\subsection{Objetivo geral}

\pendente{Um único objetivo, com verbo no infinitivo, mensurável ao fim do
TCC II. Rascunho: desenvolver e avaliar uma aplicação web de apoio à fase
preparatória de contratações públicas municipais, capaz de gerar minutas de ETP
e TR e de compor pesquisa de preços a partir de dados públicos de contratações.}

\subsection{Objetivos específicos}

\pendente{Derivar de três a cinco objetivos do geral, cada um verificável e
associado a uma entrega. Rascunho a lapidar:}

\begin{enumerate}
  \item levantar, junto a servidores do setor de licitações, os requisitos e os
    gargalos da fase preparatória;
  \item modelar a persistência dos dados de contratação, incluindo a trilha de
    auditoria que permita reproduzir uma pesquisa de preços na data em que foi
    realizada;
  \item implementar a integração com a API pública do Portal Nacional de
    Contratações Públicas e o tratamento estatístico exigido para a composição
    do valor estimado;
  \item implementar a geração assistida de minutas de ETP e TR;
  \item avaliar a solução quanto a esforço, conformidade e usabilidade.
\end{enumerate}
